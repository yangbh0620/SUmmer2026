\begin{frame}{Task 1 - Agent formulation (4 points)}
\small
Analyze a warehouse-routing agent.

\vspace{0.2cm}
\begin{enumerate}
\item[\textbf{a.}] Give one element for each PEAS component. \hfill{\color{courseblue}\textbf{[2 points]}}
\item[\textbf{b.}] Explain why rational does not mean omniscient. \hfill{\color{courseblue}\textbf{[2 points]}}
\end{enumerate}
\end{frame}

\begin{frame}{Task 2 - BFS and DFS trace (4 points)}
\small
Successors are generated alphabetically. S connects to A and B; A to G; B to C; C to G.

\vspace{0.2cm}
\begin{enumerate}
\item[\textbf{a.}] Trace BFS until a goal is removed. \hfill{\color{courseblue}\textbf{[2 points]}}
\item[\textbf{b.}] Trace DFS when successors are pushed in alphabetical order. \hfill{\color{courseblue}\textbf{[2 points]}}
\end{enumerate}
\end{frame}

\begin{frame}{Task 3 - Cost-sensitive search (6 points)}
\small
Two routes reach a goal: S-A-G costs 5+1, and S-B-C-G costs 1+1+1.

\vspace{0.2cm}
\begin{enumerate}
\item[\textbf{a.}] Which route can BFS return, and why is that not a minimum-cost guarantee? \hfill{\color{courseblue}\textbf{[2 points]}}
\item[\textbf{b.}] Trace UCS priorities until the goal is removed. \hfill{\color{courseblue}\textbf{[2 points]}}
\item[\textbf{c.}] State conditions under which UCS is complete and optimal. \hfill{\color{courseblue}\textbf{[2 points]}}
\end{enumerate}
\end{frame}

\begin{frame}{Task 4 - A-star heuristic (6 points)}
\small
A four-neighbor grid uses unit costs and Manhattan distance.

\vspace{0.2cm}
\begin{enumerate}
\item[\textbf{a.}] Define f, g, and h in A-star. \hfill{\color{courseblue}\textbf{[2 points]}}
\item[\textbf{b.}] Prove Manhattan distance is consistent. \hfill{\color{courseblue}\textbf{[2 points]}}
\item[\textbf{c.}] If two nodes have (g,h)=(4,3) and (5,1), which is removed first? \hfill{\color{courseblue}\textbf{[2 points]}}
\end{enumerate}
\end{frame}

\begin{frame}{Edge cases and defensive programming}
\small
\begin{itemize}
  \item A start state is already a goal.
  \item A graph contains a cycle.
  \item Two A-star nodes have equal f.
\end{itemize}
\end{frame}
